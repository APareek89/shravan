Daily check-in flow

%% Daily check-in flow
%% files: app/api/chat/route.ts, lib/prompts.ts, lib/safety.ts, lib/validation.ts
%% granularity: full
flowchart TD
  Request["CHAT REQUEST<br/>[FUNCTION]<br/>in: signed Elder cookie + message<br/>out: validated Elder and request"] --> PersistUser["PERSIST TURN<br/>[DATA · Postgres]<br/>in: conversation + user text<br/>out: stored message"]
  PersistUser --> Distress{"DISTRESS GATE<br/>[FUNCTION]<br/>known distress phrase?"}
  Distress -->|yes| Urgent["URGENT FAIL-SAFE<br/>[FUNCTION]<br/>in: distress text<br/>out: fixed 112 script even if alert write fails"]
  Distress -->|no| Context["LOAD CONTEXT<br/>[DATA · Postgres]<br/>in: Elder + today<br/>out: last 30 messages, meds, prior summary, usage"]
  Context --> Budget{"DAILY BUDGET<br/>[FUNCTION]<br/>tokens ≥ DAILY_TOKEN_BUDGET?"}
  Budget -->|yes| Close["SHORT SAFE CLOSE<br/>[FUNCTION]<br/>in: preferred language<br/>out: fixed closing response"]
  Budget -->|no| TurnCap{"TURN CAP<br/>[FUNCTION]<br/>all conversation user turns ≥ 25?"}
  TurnCap --> Prompt["BUILD PROMPT<br/>[FUNCTION]<br/>in: untrusted profile/history data + force-close<br/>out: data-bounded safety system prompt"]
  Prompt --> Claude["STREAM CHECK-IN<br/>[AGENT · claude-sonnet-4-6]<br/>in: prompt + ≤30 messages<br/>out: text deltas + optional save_checkin"]
  Claude --> Validate{"TOOL VALID?<br/>[LIBRARY · Zod]<br/>mood 1–5 + typed fields?"}
  Validate -->|yes| Save["UPSERT CHECK-IN<br/>[DATA · Postgres]<br/>in: one structured tool call<br/>out: one row per Elder/date"]
  Validate -->|no| Message["PERSIST ASSISTANT<br/>[DATA · Postgres]<br/>in: streamed text<br/>out: conversation continuity"]
  Save --> Message
  Message --> Result["NDJSON COMPLETE<br/>[FUNCTION]<br/>out: saved flag + token usage; failures logged"]
  Urgent --> Result
  Close --> Result

Deterministic alert and digest flow

%% Deterministic alert and digest flow
%% files: app/api/scam-check/route.ts, app/api/cron/escalations/route.ts, app/api/cron/digest/route.ts, lib/safety.ts, lib/cron.ts, scripts/call-cron.mjs, render.yaml
%% granularity: full
flowchart TD
  Cron["RENDER CRON REQUEST<br/>[FUNCTION]<br/>in: private web host + shared generated secret<br/>out: authenticated IST run"] --> Auth{"BEARER GATE<br/>[FUNCTION]<br/>exact CRON_SECRET?"}
  Auth -->|no| Deny["401<br/>[FUNCTION]<br/>out: no database work"]
  Auth -->|yes| Kind{"JOB TYPE<br/>[FUNCTION]<br/>escalation or digest?"}
  Kind -->|escalation| Time{"MISSED GATE<br/>[FUNCTION]<br/>IST hour ≥ 20 and no check-in?"}
  Time -->|yes| Missed["IDEMPOTENT ALERT<br/>[DATA · Postgres]<br/>dedupe: Elder + date + MISSED"]
  Time -->|no| Mood
  Missed --> Mood{"LOW-MOOD GATE<br/>[FUNCTION]<br/>last two consecutive dates ≤ 2?"}
  Mood -->|yes| Low["IDEMPOTENT ALERT<br/>[DATA · Postgres]<br/>dedupe: Elder + latest date + LOW"]
  Mood -->|no| Done["RUN SUMMARY<br/>[FUNCTION]<br/>out: created counts"]
  Low --> Done
  Kind -->|digest| History["WEEK DATA<br/>[DATA · Postgres]<br/>in: IST week + schedule-active days<br/>out: check-ins, adherence, alerts"]
  History --> Digest["WRITE DIGEST<br/>[AGENT · claude-haiku-4-5-20251001]<br/>in: untrusted factual JSON<br/>out: ≤150 words or logged factual fallback"]
  Digest --> Upsert["UPSERT DIGEST<br/>[DATA · Postgres]<br/>key: Elder + week_start"]
  Upsert --> Done
